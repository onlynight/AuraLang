### Fix 1：AOT `emit_call` 返回值类型推断

**问题**：`emit.rs:1039` 硬编码 `ret_ty = "i32"`，忽略函数签名返回类型

**修复文件**：`compiler/src/codegen/aot/emit.rs`

**修改方案**：

```rust
// 当前（错误）：
fn emit_call(ctx: &mut EmitCtx, blocks: &mut FuncBlocks, callee: &str, args: &[HirExpr])
    -> Result<(String, String), AotError> {
    // ...
    let ret_ty = "i32".to_string(); // 默认 i32
    // ...
}

// 修复后：
fn emit_call(ctx: &mut EmitCtx, blocks: &mut FuncBlocks, callee: &str, args: &[HirExpr])
    -> Result<(String, String), AotError> {
    // 从 HIR 函数签名推断返回类型
    let ret_ty = if let Some(func) = ctx.program.functions.iter().find(|f| f.name == callee) {
        if let Some(ref ret) = func.ret {
            ctx.types.map(ret)  // 使用 TypeMapper 映射返回类型
        } else {
            "void".to_string()  // 无返回类型
        }
    } else if let Some(native) = ctx.program.natives.iter().find(|f| f.name == callee) {
        if let Some(ref ret) = native.ret {
            ctx.types.map(ret)
        } else {
            "void".to_string()
        }
    } else {
        "i32".to_string()  // 兜底
    };
    // ...
}
```

### Fix 7：静态链接类型安全调用

**问题**：`ffi.rs:148` + `interp.rs:535-540` 所有参数强制转为 `i64`

**修复文件**：`compiler/src/vm/interp.rs` + `compiler/src/vm/ffi.rs`

```rust
// 定义函数签名描述结构
pub struct CFuncInfo {
    pub param_types: Vec<CType>,   // [Int, Float, Ptr, ...]
    pub return_type: CType,
}

enum CType {
    I64,    // i64 / int / long
    F64,    // f64 / double
    Ptr,    // void* / CString
    I32,    // int32
    Bool,   // bool / i1
}

// 类型安全调用
fn static_call_c_typed(name: &str, args: &[Value], info: &CFuncInfo) -> Option<Value> {
    let addr = resolve_static_symbol(name)?;
    // 根据 CType 转换参数
    let converted: Vec<i64> = args.iter().zip(info.param_types.iter()).map(|(v, t)| {
        match t {
            CType::F64 => (v.as_float() as i64),  // 位转换
            CType::Ptr => v.as_int(),              // 指针直接传
            CType::I64 | CType::I32 => v.as_int(),
            CType::Bool => v.as_int() & 1,
        }
    }).collect();
    // 调用（仍需 CFuncPtr，但参数已按类型转换）
    let ptr: CFuncPtr = unsafe { std::mem::transmute(addr) };
    let result = unsafe { ptr(converted[0], converted[1], converted[2], converted[3]) };
    // 返回值按类型还原
    match info.return_type {
        CType::F64 => Some(Value::Float(result as f64)),
        CType::Ptr => Some(Value::Ptr(result)),
        _ => Some(Value::Int(result)),
    }
}
```

### Fix 11：Channel 超时

pub struct Channel {
    pub id: usize,
    pub bound: usize,
    pub buffer: VecDeque<Value>,
    pub recv_timeout_ms: Option<u64>,  // 新增
}

impl Channel {
    pub fn recv_timeout(&mut self, duration_ms: u64) -> Option<Value> {
        let deadline = Instant::now() + Duration::from_millis(duration_ms);
        loop {
            if let Some(v) = self.buffer.pop_front() {
                return Some(v);
            }
            if Instant::now() >= deadline {
                return None;  // 超时
            }
            std::thread::sleep(Duration::from_millis(1));
        }
    }
}

### Fix 12：Actor 死亡传播

**问题**：`actor.rs:110-119` `kill` 仅设 `alive=false`

**修复文件**：`compiler/src/vm/actor.rs`

```rust
pub struct Actor {
    // 现有字段
    pub parent: Option<usize>,
    pub children: Vec<usize>,
    pub alive: bool,
    // 新增：监督策略
    pub supervision_policy: SupervisionPolicy,
}

pub enum SupervisionPolicy {
    Restart,      // 子 Actor 死亡时重启
    Escalate,     // 向父 Actor 报告
    Terminate,    // 终止整个监督树
}

impl ActorRuntime {
    pub fn kill(&mut self, id: usize) {
        let child = self.actors.get_mut(&id).unwrap();
        child.alive = false;
        
        // 死亡传播
        if let Some(parent_id) = child.parent {
            let parent = self.actors.get_mut(&parent_id).unwrap();
            match parent.supervision_policy {
                SupervisionPolicy::Restart => {
                    self.restart_actor(id);
                }
                SupervisionPolicy::Escalate => {
                    self.send_message(parent_id, ActorMessage::ChildDied(id));
                }
                SupervisionPolicy::Terminate => {
                    self.terminate_tree(parent_id);
                }
            }
        }
    }
}
```
